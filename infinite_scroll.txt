<template>
  <div class="container">
    <h1>Paginated Scroll Items</h1>

    <!-- 1. The main list rendering items -->
    <ul class="item-list">
      <li v-for="item in items" :key="item.id" class="list-item">
        {{ item.title }}
      </li>
    </ul>

    <!-- 2. Loading state feedback -->
    <div v-if="loading" class="loading-state">
      Loading more items...
    </div>

    <!-- 3. Target anchor element monitored by Intersection Observer -->
    <div ref="scrollTrigger" class="trigger-element"></div>
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';

// Reactive States
const items = ref([]);
const page = ref(1);
const limit = 10;
const loading = ref(false);
const hasMore = ref(true);

// Template reference for the scroll observer anchor
const scrollTrigger = ref(null);
let observer = null;

// Mock function representing your real API call
const fetchItemsFromAPI = async (currentPage, itemsLimit) => {
  try {
    // Replace with your actual endpoint URL
    const response = await fetch(`https://typicode.com{currentPage}&_limit=${itemsLimit}`);
    const data = await response.json();
    return data;
  } catch (error) {
    console.error("API error:", error);
    return [];
  }
};

// Orchestration function to manage pagination logic
const loadNextPage = async () => {
  if (loading.value || !hasMore.value) return;

  loading.value = true;
  
  // Optional delay to prevent rapid-fire requests on shaky connections
  const newItems = await fetchItemsFromAPI(page.value, limit);

  if (newItems.length < limit) {
    hasMore.value = false; // Stop observer when no more data exists
  }

  // Append new results to existing records
  items.value = [...items.value, ...newItems];
  page.value++;
  
  loading.value = false;
};

// Lifecycle: setup intersection observer when component mounts
onMounted(() => {
  observer = new IntersectionObserver((entries) => {
    // If the anchor div enters viewport, run the logic
    if (entries[0].isIntersecting && hasMore.value) {
      loadNextPage();
    }
  }, {
    rootMargin: '100px', // Pre-fetches 100px before reaching the bottom
    threshold: 0.1
  });

  if (scrollTrigger.value) {
    observer.observe(scrollTrigger.value);
  }
});

// Lifecycle: clean up to avoid memory leaks
onUnmounted(() => {
  if (observer) observer.disconnect();
});
</script>

  <script>
    const { createApp, ref, onMounted, onUnmounted } = Vue;

    createApp({
      setup() {
        const items = ref([]);
        const page = ref(1);
        const limit = ref(10);
        const loading = ref(false);
        const hasMore = ref(true);

        const fetchItems = async () => {
          if (loading.value || !hasMore.value) return;
          loading.value = true;

          try {
            const res = await fetch(`https://typicode.com{page.value}&_limit=${limit.value}`);
            const data = await res.json();

            if (data.length < limit.value) hasMore.value = false;
            items.value.push(...data);
            page.value += 1;
          } catch (err) {
            console.error(err);
          } finally {
            loading.value = false;
          }
        };

        const handleScroll = () => {
          const bottomOfWindow = window.innerHeight + window.scrollY >= document.documentElement.scrollHeight - 50;
          if (bottomOfWindow) {
            fetchItems();
          }
        };

        onMounted(() => {
          fetchItems();
          window.addEventListener('scroll', handleScroll);
        });

        onUnmounted(() => {
          window.removeEventListener('scroll', handleScroll);
        });

        return { items, loading };
      }
    }).mount('#app');
  </script>

        <h3>{{ group.letter }}</h3>
        <div v-for="item in group.items" :key="item" class="item-row">
          {{ item }}
        </div>
      </div>
      <p v-if="loading" style="text-align:center;">Loading more...</p>
    </div>

    <!-- Alphabet Section Selector Sidebar -->
    <div class="sidebar-alpha">
      <button 
        v-for="char in alphabet" 
        :key="char" 
        @click="scrollToLetter(char)"
        :class="{ active: activeLetter === char }">
        {{ char }}
      </button>
    </div>
  </div>

  <script>
    const { createApp, ref, computed, onMounted } = Vue;

    createApp({
      setup() {
        const alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split("");
        const activeLetter = ref("A");
        const loading = ref(false);
        const container = ref(null);
        
        // Generate mock dataset categorized by letters
        const rawData = {};
        alphabet.forEach(letter => {
          rawData[letter] = Array.from({ length: 15 }, (_, i) => `${letter} - Item Example ${i + 1}`);
        });

        // Limit initial display count for scrolling pagination chunks
        const limitCount = ref(5);
        const visibleGroups = computed(() => {
          return alphabet.slice(0, limitCount.value).map(letter => ({
            letter,
            items: rawData[letter]
          }));
        });

        // Infinite scroll loader simulation
        const handleScroll = (e) => {
          const el = e.target;
          if (el.scrollTop + el.clientHeight >= el.scrollHeight - 20) {
            if (limitCount.value < alphabet.length && !loading.value) {
              loading.value = true;
              setTimeout(() => {
                limitCount.value = Math.min(limitCount.value + 3, alphabet.length);
                loading.value = false;
              }, 600);
            }
          }
        };

        // Jump or scroll smoothly to the target alphabet section
        const scrollToLetter = (char) => {
          activeLetter.value = char;
          // Expand list limit if letter isn't loaded yet
          const charIndex = alphabet.indexOf(char);
          if (charIndex >= limitCount.value) {
            limitCount.value = charIndex + 1;
          }
          setTimeout(() => {
            const el = document.getElementById('sec-' + char);
            if (el && container.value) {
              container.value.scrollTo({ top: el.offsetTop - container.value.offsetTop, behavior: 'smooth' });
            }
          }, 100);
        };

        return {
          alphabet,
          activeLetter,
          visibleGroups,
          loading,
          container,
          handleScroll,
          scrollToLetter
        };
      }
    }).mount('#app');
  </script>
</body>
